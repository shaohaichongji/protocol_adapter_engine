cmake_minimum_required(VERSION 3.20)
project(pae_config_tutorial LANGUAGES CXX)
find_package(PAE CONFIG REQUIRED)
add_executable(config_tutorial verify.cpp)
target_compile_features(config_tutorial PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(config_tutorial PRIVATE /utf-8)
  set_property(TARGET config_tutorial PROPERTY PAE_MSVC_SOURCE_CHARSET_SUPPLIED ON)
endif()
target_link_libraries(config_tutorial PRIVATE PAE::pae)
enable_testing()
add_test(NAME tutorial_binary COMMAND config_tutorial "${CMAKE_CURRENT_SOURCE_DIR}/01-最小二进制.pae.json" binary)
add_test(NAME tutorial_ascii COMMAND config_tutorial "${CMAKE_CURRENT_SOURCE_DIR}/02-最小文本.pae.json" ascii)
add_test(NAME tutorial_missing_property COMMAND config_tutorial "${CMAKE_CURRENT_SOURCE_DIR}/01-最小二进制.pae.json" missing-property)
add_test(NAME tutorial_unknown_reference COMMAND config_tutorial "${CMAKE_CURRENT_SOURCE_DIR}/01-最小二进制.pae.json" unknown-reference)
