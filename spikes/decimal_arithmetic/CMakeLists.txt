cmake_minimum_required(VERSION 3.25...4.4)
project(PaeDecimalArithmeticSpike LANGUAGES CXX)
# Deliberately standalone: no add_subdirectory from the product root, install, or export.
enable_testing()
add_executable(pae_decimal_arithmetic_spike arithmetic_tests.cpp)
target_compile_features(pae_decimal_arithmetic_spike PRIVATE cxx_std_17)
set_target_properties(pae_decimal_arithmetic_spike PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(pae_decimal_arithmetic_spike PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(pae_decimal_arithmetic_spike PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME pae.spike.decimal_arithmetic COMMAND pae_decimal_arithmetic_spike)
set_tests_properties(pae.spike.decimal_arithmetic PROPERTIES TIMEOUT 120)
