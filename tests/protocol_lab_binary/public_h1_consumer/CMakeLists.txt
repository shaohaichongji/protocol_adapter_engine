cmake_minimum_required(VERSION 3.25)
project(PublicBinaryH1Consumer LANGUAGES CXX)
find_package(PAE CONFIG REQUIRED)
if(NOT H1_ADAPTER_DIR)
  message(FATAL_ERROR "H1_ADAPTER_DIR must name only the public H1 adapter sources.")
endif()
add_executable(h1_consumer main.cpp "${H1_ADAPTER_DIR}/public_binary_decode.cpp")
target_include_directories(h1_consumer PRIVATE "${H1_ADAPTER_DIR}")
target_link_libraries(h1_consumer PRIVATE PAE::pae)
target_compile_features(h1_consumer PRIVATE cxx_std_17)
